# 简单的 Redis 任务入队示例
# 任务格式（JSON）示例：
# {
#   "id": "task-20260108-0001",
#   "input": "C:/images/input1.jpg",
#   "output": "C:/images/out1.jpg",
#   "action": {"type": "photoshop_action", "set": "MySet", "name": "Oilify"},
#   "retries": 3
# }

import json
import redis
import uuid

r = redis.Redis(host='localhost', port=6379, db=0, decode_responses=True)
QUEUE_KEY = 'photoshop:tasks'

def push_task(task: dict):
    task['id'] = task.get('id') or str(uuid.uuid4())
    r.rpush(QUEUE_KEY, json.dumps(task))
    return task['id']

if __name__ == '__main__':
    example = {
        'input': r'C:\images\input1.jpg',
        'output': r'C:\images\out1.jpg',
        'action': {'type': 'photoshop_action', 'set': 'MySet', 'name': 'Oilify'},
        'retries': 3
    }
    tid = push_task(example)
    print('pushed', tid)
"""

worker_py = r"""
# Photoshop Worker
# 功能：
#  1) 从 Redis 或 文件队列 拉取任务
#  2) 生成临时 JSX 文件（含用户传入的 action 或 jsx_snippet）
#  3) 调用 Photoshop 执行 JSX（Windows 使用 COM；macOS 使用 osascript）
#  4) 支持超时/重试/日志

import os
import json
import time
import tempfile
import logging
import logging.handlers
import shutil
import traceback
from pathlib import Path
from multiprocessing import Process

# 如果使用 Redis，取消注释下面并安装 redis
try:
    import redis
except Exception:
    redis = None

# Windows COM
try:
    import win32com.client
    from win32com.client import Dispatch
except Exception:
    win32com = None

# config
USE_REDIS = True if redis is not None else False
REDIS_QUEUE_KEY = 'photoshop:tasks'
REDIS_CONFIG = {'host': 'localhost', 'port': 6379, 'db': 0}
FILE_QUEUE_DIR = Path('./file_queue')
FILE_QUEUE_DIR.mkdir(exist_ok=True)

LOG_PATH = Path('./logs')
LOG_PATH.mkdir(exist_ok=True)

logger = logging.getLogger('ps_worker')
logger.setLevel(logging.INFO)
handler = logging.handlers.RotatingFileHandler(LOG_PATH / 'worker.log', maxBytes=5_000_000, backupCount=3, encoding='utf-8')
fmt = logging.Formatter('%(asctime)s [%(levelname)s] %(message)s')
handler.setFormatter(fmt)
logger.addHandler(handler)

# --------------------- JSX 模板 ---------------------
# 说明：MIDDLE_ACTION 会被替换为 task['action'] 的处理内容；
# 如果 action.type == 'photoshop_action', 建议用 app.doAction('ActionName','ActionSet');

JSX_TEMPLATE = r'''
// Auto-generated JSX by python worker
#target photoshop

var inputPath = "%INPUT_PATH%";
var outputPath = "%OUTPUT_PATH%";

function main(){
    try {
        var inFile = new File(inputPath);
        if(!inFile.exists){
            throw('Input file does not exist: ' + inputPath);
        }
        var doc = app.open(inFile);

        // === MIDDLE_ACTION_START ===
        %MIDDLE_ACTION%
        // === MIDDLE_ACTION_END ===

        // Save as JPEG with quality 10 (示例，可改)
        var saveFile = new File(outputPath);
        var opts = new JPEGSaveOptions();
        opts.quality = 10;
        doc.saveAs(saveFile, opts, true);
        doc.close(SaveOptions.DONOTSAVECHANGES);
    } catch(e){
        if(app.documents.length > 0) { try { app.activeDocument.close(SaveOptions.DONOTSAVECHANGES); } catch(e2){} }
        throw(e);
    }
}

main();
'''

# --------------------- Photoshop 调用（Windows / macOS） ---------------------

def call_photoshop_jsx_windows(jsx_path, timeout_seconds=60):
    """在子进程中通过 COM 调用 Photoshop 执行 JSX（Windows）。
    通过 multiprocessing 启动单独进程以便可以实施超时控制。
    """
    def target(path):
        try:
            # 通过 COM 连接到 Photoshop
            app = Dispatch('Photoshop.Application')
            # 提供可见性（可选）
            app.Visible = True
            # DoJavaScriptFile 或 DoJavaScript
            # 有的版本是 app.DoJavaScriptFile
            try:
                app.DoJavaScriptFile(path)
            except Exception:
                with open(path, 'r', encoding='utf-8') as f:
                    code = f.read()
                app.DoJavaScript(code)
        except Exception:
            # 将错误写入临时日志，主进程会读取
            raise

    p = Process(target=target, args=(jsx_path,))
    p.start()
    p.join(timeout=timeout_seconds)
    if p.is_alive():
        logger.warning('Photoshop jsx execution timeout, terminating process')
        p.terminate()
        p.join(5)
        raise TimeoutError('Photoshop jsx execution timed out')
    if p.exitcode != 0:
        raise RuntimeError(f'Photoshop process exited with code {p.exitcode}')


def call_photoshop_jsx_macos(jsx_path, timeout_seconds=60):
    """macOS: 使用 osascript 调用 Photoshop 执行 JavaScript 文件。
    """
    import subprocess
    # 应用名示例："Adobe Photoshop 2023"，按你系统上的应用名修改
    app_name = 'Adobe Photoshop 2023'
    cmd = [
        'osascript', '-e', f'tell application "{app_name}" to do javascript (POSIX file "{jsx_path}")'
    ]
    try:
        res = subprocess.run(cmd, timeout=timeout_seconds, capture_output=True)
        if res.returncode != 0:
            raise RuntimeError(res.stderr.decode('utf-8', errors='ignore'))
    except subprocess.TimeoutExpired:
        raise TimeoutError('Photoshop jsx execution timed out (osascript)')

# --------------------- 队列拉取（Redis / File） ---------------------

def pop_task_redis(redis_client, block_timeout=5):
    # BRPOP blocks for block_timeout seconds
    res = redis_client.brpop(REDIS_QUEUE_KEY, timeout=block_timeout)
    if not res:
        return None
    _key, raw = res
    return json.loads(raw)


def pop_task_file():
    # 以 FIFO 读取目录内最早的文件
    files = sorted(FILE_QUEUE_DIR.glob('*.json'), key=lambda p: p.stat().st_mtime)
    if not files:
        return None
    p = files[0]
    try:
        with open(p, 'r', encoding='utf-8') as f:
            task = json.load(f)
        # 先移动文件到处理目录，避免重复处理
        processing = FILE_QUEUE_DIR / 'processing'
        processing.mkdir(exist_ok=True)
        dest = processing / p.name
        shutil.move(str(p), str(dest))
        task['_file_path'] = str(dest)
        return task
    except Exception:
        logger.exception('Failed to read task file %s', p)
        return None

# --------------------- Worker 主循环 ---------------------

def run_worker(max_idle_seconds=300):
    logger.info('Worker started. USE_REDIS=%s', USE_REDIS)
    redis_client = None
    if USE_REDIS:
        redis_client = redis.Redis(**REDIS_CONFIG)

    idle_start = time.time()

    while True:
        try:
            task = None
            if USE_REDIS and redis_client:
                task = pop_task_redis(redis_client, block_timeout=5)
            else:
                task = pop_task_file()

            if not task:
                # idle handling
                if time.time() - idle_start > max_idle_seconds:
                    logger.info('Worker idle timeout, exiting')
                    break
                time.sleep(1)
                continue

            idle_start = time.time()
            logger.info('Got task %s', task.get('id'))
            process_task(task)
        except Exception as e:
            logger.exception('Unhandled error in main loop: %s', e)
            time.sleep(2)


def process_task(task: dict):
    tid = task.get('id', 'unknown')
    retries_left = int(task.get('retries', 0))
    attempt = 0
    max_attempts = retries_left + 1
    backoff_base = 2

    while attempt < max_attempts:
        attempt += 1
        try:
            logger.info('Task %s attempt %d/%d', tid, attempt, max_attempts)
            do_execute_task(task)
            logger.info('Task %s finished', tid)
            # 如果是文件队列，处理完删除文件
            if '_file_path' in task:
                try:
                    os.remove(task['_file_path'])
                except Exception:
                    logger.exception('Failed to remove task file')
            return
        except TimeoutError as te:
            logger.warning('Task %s timeout: %s', tid, te)
            # 超时一般重试
        except Exception as e:
            logger.exception('Task %s failed: %s', tid, e)

        # 重试前等待（指数退避）
        sleep_seconds = backoff_base ** attempt
        logger.info('Task %s will retry after %ds', tid, sleep_seconds)
        time.sleep(sleep_seconds)

    logger.error('Task %s exhausted retries', tid)


def do_execute_task(task: dict):
    # 1) 生成 JSX 文件
    inp = task.get('input')
    outp = task.get('output')
    if not inp or not outp:
        raise ValueError('task must have input and output')

    middle_action_code = ''
    action = task.get('action') or {}
    if action.get('type') == 'photoshop_action':
        # 使用 app.doAction('name','set')
        set_name = action.get('set', '')
        act_name = action.get('name', '')
        middle_action_code = f"app.doAction('{act_name}','{set_name}');"
    elif action.get('type') == 'jsx_snippet':
        middle_action_code = action.get('code', '')
    elif action.get('type') == 'auto_exposure_example':
        # 这是示例占位；推荐改为真实 ExtendScript 或 Photoshop Action
        middle_action_code = "// 将在这里插入自动曝光/处理的 ExtendScript 代码\n" + action.get('code', '')
    else:
        # 默认：提示用户用 action 或 snippet
        middle_action_code = "// NO ACTION PROVIDED - please supply an action or jsx_snippet"

    # 确保路径反斜杠转义
    inp_esc = inp.replace('\\', '\\\\')
    out_esc = outp.replace('\\', '\\\\')

    jsx_content = JSX_TEMPLATE.replace('%INPUT_PATH%', inp_esc).replace('%OUTPUT_PATH%', out_esc).replace('%MIDDLE_ACTION%', middle_action_code)

    tmp = tempfile.NamedTemporaryFile(delete=False, suffix='.jsx', prefix='ps_task_')
    tmp_path = tmp.name
    tmp.write(jsx_content.encode('utf-8'))
    tmp.close()

    logger.info('Generated temporary jsx %s for task %s', tmp_path, task.get('id'))

    # 2) 调用 Photoshop 执行 JSX
    timeout = int(task.get('timeout_seconds', 120))

    try:
        if os.name == 'nt' and win32com is not None:
            call_photoshop_jsx_windows(tmp_path, timeout_seconds=timeout)
        elif os.name == 'posix':
            call_photoshop_jsx_macos(tmp_path, timeout_seconds=timeout)
        else:
            raise RuntimeError('Unsupported OS or missing Photoshop interface')
    finally:
        # 清理临时文件（可选保留以作调试）
        try:
            os.remove(tmp_path)
        except Exception:
            pass

# --------------------- 启动 ---------------------

if __name__ == '__main__':
    try:
        run_worker()
    except KeyboardInterrupt:
        logger.info('Worker interrupted, exiting')
"""
